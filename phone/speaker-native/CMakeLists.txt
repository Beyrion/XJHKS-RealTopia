cmake_minimum_required(VERSION 3.22)
project(realtopia_speaker_mnn LANGUAGES C CXX)

if(NOT ANDROID OR NOT DEFINED MNN_ANDROID_LIB_DIR)
    message(FATAL_ERROR "Android and MNN_ANDROID_LIB_DIR are required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "Dependency compatibility with CMake 4")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(MNN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../MNN" ABSOLUTE)
set(SHERPA_ROOT "${MNN_ROOT}/apps/frameworks/sherpa-mnn")
add_library(MNN SHARED IMPORTED)
set_target_properties(MNN PROPERTIES IMPORTED_LOCATION "${MNN_ANDROID_LIB_DIR}/libMNN.so")
add_library(MNN_Express SHARED IMPORTED)
set_target_properties(MNN_Express PROPERTIES IMPORTED_LOCATION "${MNN_ANDROID_LIB_DIR}/libMNN_Express.so")
set(MNN_LIB_DIR "${MNN_ROOT}")
include_directories("${SHERPA_ROOT}" "${MNN_ROOT}/include")
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(SHERPA_MNN_ENABLE_JNI ON CACHE BOOL "" FORCE)
set(SHERPA_MNN_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SHERPA_MNN_ENABLE_BINARY OFF CACHE BOOL "" FORCE)
set(SHERPA_MNN_ENABLE_TTS OFF CACHE BOOL "" FORCE)
set(SHERPA_MNN_ENABLE_PORTAUDIO OFF CACHE BOOL "" FORCE)
set(SHERPA_MNN_ENABLE_WEBSOCKET OFF CACHE BOOL "" FORCE)
set(SHERPA_MNN_ENABLE_SPEAKER_DIARIZATION ON CACHE BOOL "" FORCE)
set(SHERPA_MNN_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SHERPA_ROOT}" sherpa EXCLUDE_FROM_ALL)

# Only expose speaker/stream JNI. No ONNX Runtime is configured or linked;
# the upstream sherpa-mnn model implementations use MNN::Express::Module.
set_target_properties(sherpa-mnn-jni PROPERTIES SOURCES
    "${SHERPA_ROOT}/sherpa-mnn/jni/jni.cc;${SHERPA_ROOT}/sherpa-mnn/jni/online-stream.cc;${SHERPA_ROOT}/sherpa-mnn/jni/speaker-embedding-extractor.cc;${SHERPA_ROOT}/sherpa-mnn/jni/speaker-embedding-manager.cc;${SHERPA_ROOT}/sherpa-mnn/jni/offline-speaker-diarization.cc")
target_link_libraries(sherpa-mnn-core MNN_Express)
target_compile_options(sherpa-mnn-core PRIVATE -ffunction-sections -fdata-sections)
target_link_options(sherpa-mnn-jni PRIVATE -Wl,--gc-sections)
